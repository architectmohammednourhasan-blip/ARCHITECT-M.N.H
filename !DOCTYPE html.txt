<script>
let projectCount = 0;
let projects = [];
let currentLightboxIndex = 0;

window.onload = function() {
    const saved = localStorage.getItem('portfolioProjects');
    if(saved){
        projects = JSON.parse(saved);
        projects.forEach(p => addProject(p.image,p.description,false));
    }
}

document.getElementById('projectFile').addEventListener('change', function(event) {
    const file = event.target.files[0];
    if(!file) return;
    const reader = new FileReader();
    reader.onload = function(e) {
        const imageData = e.target.result;
        const description = prompt("أدخل وصف المشروع");
        if(!description) return;
        addProject(imageData, description,true);
    };
    reader.readAsDataURL(file);
    event.target.value = '';
});

function addProject(imageUrl, description, save=true){
    projectCount++;
    const portfolio = document.getElementById('portfolio');

    const projectDiv = document.createElement('div');
    projectDiv.className = 'portfolio-item';
    projectDiv.setAttribute('draggable', true);
    projectDiv.innerHTML = `
        <img src="${imageUrl}" alt="Project ${projectCount}" onclick="openLightbox(${projects.length})">
        <div class="portfolio-overlay" onclick="editDescription(this)">${description}</div>
    `;
    portfolio.appendChild(projectDiv);

    enableDragDrop();

    if(save){
        projects.push({image:imageUrl,description:description});
        saveProjects();
    }
}

function deleteProject(){
    if(projectCount===0) return;
    const portfolio=document.getElementById('portfolio');
    portfolio.removeChild(portfolio.lastElementChild);
    projects.pop();
    saveProjects();
    projectCount--;
}

// Lightbox مع تأثير ضبابية الخلفية
function openLightbox(index){
    currentLightboxIndex = index;
    document.getElementById('lightboxImage').src = projects[index].image;
    document.getElementById('lightboxContainer').classList.add('active');
    document.body.classList.add('blur');
}

function closeLightbox(){
    document.getElementById('lightboxContainer').classList.remove('active');
    document.body.classList.remove('blur');
}

function nextProject(){
    currentLightboxIndex = (currentLightboxIndex + 1) % projects.length;
    document.getElementById('lightboxImage').src = projects[currentLightboxIndex].image;
}

function prevProject(){
    currentLightboxIndex = (currentLightboxIndex - 1 + projects.length) % projects.length;
    document.getElementById('lightboxImage').src = projects[currentLightboxIndex].image;
}

function editDescription(overlay){
    const newDesc = prompt("عدل وصف المشروع:", overlay.textContent);
    if(newDesc !== null){
        overlay.textContent = newDesc;
        const index = Array.from(document.getElementById('portfolio').children).indexOf(overlay.parentElement);
        projects[index].description = newDesc;
        saveProjects();
    }
}

function saveProjects(){localStorage.setItem('portfolioProjects',JSON.stringify(projects));}

// السحب والإفلات
function enableDragDrop(){
    const items = document.querySelectorAll('.portfolio-item');
    items.forEach(item => {
        item.addEventListener('dragstart', dragStart);
        item.addEventListener('dragover', dragOver);
        item.addEventListener('drop', drop);
        item.addEventListener('dragend', dragEnd);
    });
}
let draggedItem=null;
function dragStart(e){draggedItem=this; setTimeout(()=>this.style.display='none',0);}
function dragOver(e){e.preventDefault();}
function drop(){
    if(this!==draggedItem){
        const portfolio=this.parentNode;
        portfolio.insertBefore(draggedItem,this);
        updateProjectsOrder();
    }
}
function dragEnd(){this.style.display='block'; draggedItem=null;}

function updateProjectsOrder(){
    const portfolio=document.getElementById('portfolio');
    const items=portfolio.children;
    const newOrder=[];
    for(let i=0;i<items.length;i++){
        const index = Array.from(document.getElementById('portfolio').children).indexOf(items[i]);
        newOrder.push(projects[index]);
    }
    projects = newOrder;
    saveProjects();
}
